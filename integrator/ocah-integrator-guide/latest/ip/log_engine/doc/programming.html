Programming
Simple Log Transfer
// Initialize log engine for 4KB region at 0x10000000
// UART at 0x40001000

#define LOG_ENGINE_BASE 0x50000000
#define LOG_REGION_ADDR 0x10000000
#define LOG_REGION_SIZE 0x1000
#define UART_TX_ADDR    0x40001000

// Disable and configure
write32(LOG_ENGINE_BASE + 0x00, 0);          // Disable
write32(LOG_ENGINE_BASE + 0x04, LOG_REGION_SIZE);
write32(LOG_ENGINE_BASE + 0x08, LOG_REGION_ADDR);
write32(LOG_ENGINE_BASE + 0x0C, 0);          // High address
write32(LOG_ENGINE_BASE + 0x10, UART_TX_ADDR);
write32(LOG_ENGINE_BASE + 0x18, 0x11);       // Enable error interrupts
write32(LOG_ENGINE_BASE + 0x00, 1);          // Enable

// Send 100-byte log via entry 0
write32(LOG_ENGINE_BASE + 0x40, 100);

// Wait for completion
while (read32(LOG_ENGINE_BASE + 0x40) != 0);
Multiple Entry Management
// Send logs via multiple entries simultaneously
write32(LOG_ENGINE_BASE + 0x40, 50);   // Entry 0: 50 bytes
write32(LOG_ENGINE_BASE + 0x44, 75);   // Entry 1: 75 bytes
write32(LOG_ENGINE_BASE + 0x48, 120);  // Entry 2: 120 bytes

// Check completion status
for (int i = 0; i < 3; i++) {
    uint32_t len = read32(LOG_ENGINE_BASE + 0x40 + (i * 4));
    if (len == 0) {
        printf("Entry %d completed\n", i);
    } else {
        printf("Entry %d: %d bytes remaining\n", i, len);
    }
}
